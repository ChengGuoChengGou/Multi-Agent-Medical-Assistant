// ============================================================ // 10. SSE Streaming Chat Handler // ============================================================ async function handleStreamChat(message, thinkingElement, agentStatus) { const chatBody = document.getElementById('chat-body'); const sendBtn = document.getElementById('send-btn'); // Create assistant message element with empty content const assistantMessageElement = document.createElement('div'); assistantMessageElement.className = 'message assistant-message'; assistantMessageElement.innerHTML = ` STREAMING
▊
`; let fullText = ''; let agentName = ''; try { const response = await fetch('/chat/stream', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ query: message, conversation_history: [] }), credentials: 'include' }); if (!response.ok) { throw new Error(`HTTP ${response.status}: ${response.statusText}`); } const reader = response.body.getReader(); const decoder = new TextDecoder(); let buffer = ''; // Remove thinking indicator and show message chatBody.removeChild(thinkingElement); chatBody.appendChild(assistantMessageElement); const contentDiv = assistantMessageElement.querySelector('.message-content'); while (true) { const { done, value } = await reader.read(); if (done) break; buffer += decoder.decode(value, { stream: true }); const lines = buffer.split('\n'); buffer = lines.pop() || ''; for (const line of lines) { if (!line.startsWith('data: ')) continue; const jsonStr = line.slice(6); try { const event = JSON.parse(jsonStr); if (event.type === 'token') { fullText += event.content; contentDiv.innerHTML = marked.parse(fullText) + '▊'; chatBody.scrollTop = chatBody.scrollHeight; } else if (event.type === 'agent') { agentName = event.content; assistantMessageElement.querySelector('.message-agent-tag').textContent = agentName; } else if (event.type === 'done') { // Finalize: remove cursor, show clean content contentDiv.innerHTML = marked.parse(fullText || event.full_response || ''); agentName = agentName || event.agent || 'ASSISTANT'; assistantMessageElement.querySelector('.message-agent-tag').textContent = agentName; // Show audio controls const audioCtrl = assistantMessageElement.querySelector('.audio-controls'); if (audioCtrl) audioCtrl.style.display = ''; } else if (event.type === 'error') { contentDiv.innerHTML = `⚠️ Error: ${event.message}`; } } catch (parseErr) { // Skip non-JSON lines } } } } catch (fetchErr) { // Remove thinking if still present if (thinkingElement.parentNode) chatBody.removeChild(thinkingElement); if (!fullText) { assistantMessageElement.innerHTML = `
⚠️ Stream error: ${fetchErr.message}
`; chatBody.appendChild(assistantMessageElement); } } finally { if (agentStatus) agentStatus.style.display = 'none'; if (sendBtn) sendBtn.disabled = false; // Add TTS click handler to play button const playButton = assistantMessageElement.querySelector('.play-tts-btn'); if (playButton && fullText) { const cleanText = removeMarkdown(fullText); playButton.addEventListener('click', async function() { const textForSpeech = cleanText.length > 1000 ? cleanText.substring(0, 1000) + '...' : cleanText; this.disabled = true; this.innerHTML = ' Generating...'; try { const resp = await fetch('/generate-speech', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ text: textForSpeech }), credentials: 'include' }); if (!resp.ok) throw new Error('TTS failed'); const audioBlob = await resp.blob(); const audioUrl = URL.createObjectURL(audioBlob); const audio = new Audio(audioUrl); audio.play(); this.innerHTML = ' Playing...'; audio.onended = () => { this.innerHTML = ' Play Voice'; this.disabled = false; URL.revokeObjectURL(audioUrl); }; } catch (err) { this.innerHTML = ' Play Voice'; this.disabled = false; } }); } } } // Add CSS animation for cursor blink const streamStyle = document.createElement('style'); streamStyle.textContent = '@keyframes blink { 0%, 100% { opacity: 1; } 50% { opacity: 0; } }'; document.head.appendChild(streamStyle);
